⚙️ Configuration Supabase

Connectez votre base de données. À faire une seule fois.

1
Créer un projet Supabase gratuit

Allez sur supabase.com → New Project → région Europe West.

2
Créer les tables (SQL Editor)

Dans Supabase → SQL Editor → copiez-collez ce script :

-- ═══════════════════════════════════════════ -- MIGRATION SQL ExamPlatform — Multi-Admin -- Exécuter dans Supabase → SQL Editor -- ═══════════════════════════════════════════ -- 1. Table admins CREATE TABLE IF NOT EXISTS admins ( id uuid PRIMARY KEY DEFAULT gen_random_uuid(), identifiant text UNIQUE NOT NULL, nom text DEFAULT '', email text DEFAULT '', pwd text NOT NULL, role text DEFAULT 'enseignant', valide boolean DEFAULT false, created_at timestamptz DEFAULT now() ); -- 2. Super admin (pwd: superadmin2024 — changez-le!) INSERT INTO admins (identifiant, nom, email, pwd, role, valide) VALUES ('admin', 'Super Administrateur', 'admin@examplatform.local', 'superadmin2024', 'superadmin', true) ON CONFLICT (identifiant) DO NOTHING; -- 3. Colonnes admin_id sur tables existantes ALTER TABLE exam_config ADD COLUMN IF NOT EXISTS admin_id uuid REFERENCES admins(id); ALTER TABLE exam_config ADD COLUMN IF NOT EXISTS universite text DEFAULT ''; ALTER TABLE exam_config ADD COLUMN IF NOT EXISTS entite text DEFAULT ''; ALTER TABLE exam_config ADD COLUMN IF NOT EXISTS logo_universite text DEFAULT ''; ALTER TABLE exam_config ADD COLUMN IF NOT EXISTS logo_entite text DEFAULT ''; ALTER TABLE exam_config ADD COLUMN IF NOT EXISTS resend_key text DEFAULT ''; ALTER TABLE exam_config ADD COLUMN IF NOT EXISTS smtp_from text DEFAULT ''; ALTER TABLE exam_config ADD COLUMN IF NOT EXISTS email_verification boolean DEFAULT false; ALTER TABLE candidats ADD COLUMN IF NOT EXISTS admin_id uuid; ALTER TABLE questions ADD COLUMN IF NOT EXISTS admin_id uuid; ALTER TABLE copies ADD COLUMN IF NOT EXISTS admin_id uuid; -- 4. Nouvelles tables CREATE TABLE IF NOT EXISTS tokens_composition ( token uuid PRIMARY KEY DEFAULT gen_random_uuid(), matricule text NOT NULL, admin_id uuid REFERENCES admins(id), expires_at timestamptz NOT NULL, used boolean DEFAULT false, created_at timestamptz DEFAULT now() ); CREATE TABLE IF NOT EXISTS reset_tokens ( token text PRIMARY KEY, admin_id uuid REFERENCES admins(id), expires_at timestamptz NOT NULL, used boolean DEFAULT false, created_at timestamptz DEFAULT now() ); CREATE TABLE IF NOT EXISTS universites ( id uuid PRIMARY KEY DEFAULT gen_random_uuid(), nom text NOT NULL, entites jsonb DEFAULT '[]', created_at timestamptz DEFAULT now() ); CREATE TABLE IF NOT EXISTS niveaux ( id uuid PRIMARY KEY DEFAULT gen_random_uuid(), nom text NOT NULL, ordre integer DEFAULT 0 ); -- 5. Niveaux par défaut INSERT INTO niveaux (nom, ordre) VALUES ('Licence 1',1),('Licence 2',2),('Licence 3',3), ('Master 1',4),('Master 2',5),('Doctorat',6) ON CONFLICT DO NOTHING; -- 6. RLS policies ALTER TABLE admins ENABLE ROW LEVEL SECURITY; ALTER TABLE tokens_composition ENABLE ROW LEVEL SECURITY; ALTER TABLE reset_tokens ENABLE ROW LEVEL SECURITY; ALTER TABLE universites ENABLE ROW LEVEL SECURITY; ALTER TABLE niveaux ENABLE ROW LEVEL SECURITY; CREATE POLICY "public_all_admins" ON admins FOR ALL USING (true) WITH CHECK (true); CREATE POLICY "public_all_tokens" ON tokens_composition FOR ALL USING (true) WITH CHECK (true); CREATE POLICY "public_all_reset" ON reset_tokens FOR ALL USING (true) WITH CHECK (true); CREATE POLICY "public_all_universites" ON universites FOR ALL USING (true) WITH CHECK (true); CREATE POLICY "public_all_niveaux" ON niveaux FOR ALL USING (true) WITH CHECK (true); -- 7. Créer le bucket Storage "logos" (public) dans Supabase Dashboard → Storage
